using System.CommandLine;
using System.Text;

var rootCommand = new RootCommand("Code Bundler CLI");

// =====================
// BUNDLE COMMAND
// =====================

var bundleCommand = new Command("bundle", "Bundle code files into a single file");

// ---------- OPTIONS ----------

var languageOption = new Option<string[]>(
    name: "--language",
    description: "Programming languages to include (or 'all')")
{
    IsRequired = true,
    AllowMultipleArgumentsPerToken = true
};
languageOption.AddAlias("-l");

// output 
var outputOption = new Option<FileInfo>(
    name: "--output",
    description: "Output file path and name")
{
    IsRequired = true
};
outputOption.AddAlias("-o");

// note
var noteOption = new Option<bool>(
    name: "--note",
    description: "Add source file note");
noteOption.AddAlias("-n");

// sort
var sortOption = new Option<string>(
    name: "--sort",
    getDefaultValue: () => "name",
    description: "Sort by 'name' or 'type'");
sortOption.AddAlias("-s");

// remove empty lines
var removeEmptyOption = new Option<bool>(
    name: "--remove-empty-lines",
    description: "Remove empty lines");
removeEmptyOption.AddAlias("-r");

// author
var authorOption = new Option<string?>(
    name: "--author",
    description: "Author name");
authorOption.AddAlias("-a");

// Add options
bundleCommand.AddOption(languageOption);
bundleCommand.AddOption(outputOption);
bundleCommand.AddOption(noteOption);
bundleCommand.AddOption(sortOption);
bundleCommand.AddOption(removeEmptyOption);
bundleCommand.AddOption(authorOption);

// ---------- HANDLER ----------

bundleCommand.SetHandler((languages, output, note, sort, removeEmpty, author) =>
{
    try
    {
        var currentDir = Directory.GetCurrentDirectory();

        var allowedExtensions = new Dictionary<string, string[]>
        {
            { "csharp", new[] { ".cs" } },
            { "javascript", new[] { ".js" } },
            { "html", new[] { ".html", ".htm" } },
            { "css", new[] { ".css" } },
                { "text", new[] { ".txt" } } 

        };

        IEnumerable<string> files = Directory.GetFiles(currentDir, "*.*", SearchOption.AllDirectories)
            .Where(f => !f.Contains("\\bin\\") &&
                        !f.Contains("\\obj\\") &&
                        !f.Contains("\\debug\\"));

        if (!languages.Contains("all"))
        {
            var extensions = languages
                .Where(l => allowedExtensions.ContainsKey(l.ToLower()))
                .SelectMany(l => allowedExtensions[l.ToLower()])
                .ToList();

            if (!extensions.Any())
            {
                Console.WriteLine("Invalid language selection.");
                return;
            }

            files = files.Where(f => extensions.Contains(Path.GetExtension(f)));
        }

        // Sorting
        files = sort.ToLower() switch
        {
            "type" => files.OrderBy(f => Path.GetExtension(f)),
            "name" => files.OrderBy(f => Path.GetFileName(f)),
            _ => files.OrderBy(f => Path.GetFileName(f))
        };

        var sb = new StringBuilder();

        // Author
        if (!string.IsNullOrWhiteSpace(author))
        {
            sb.AppendLine($"// Author: {author}");
            sb.AppendLine($"// Created: {DateTime.Now}");
            sb.AppendLine();
        }

        foreach (var file in files)
        {
            if (note)
            {
                var relativePath = Path.GetRelativePath(currentDir, file);
                sb.AppendLine($"// Source: {relativePath}");
            }

            var lines = File.ReadAllLines(file);

            if (removeEmpty)
                lines = lines.Where(l => !string.IsNullOrWhiteSpace(l)).ToArray();

            foreach (var line in lines)
                sb.AppendLine(line);

            sb.AppendLine();
        }

        File.WriteAllText(output.FullName, sb.ToString());

        Console.WriteLine("Bundle created successfully!");
    }
    catch (Exception ex)
    {
        Console.WriteLine($"Error: {ex.Message}");
    }

}, languageOption, outputOption, noteOption, sortOption, removeEmptyOption, authorOption);

rootCommand.AddCommand(bundleCommand);

// =====================
// CREATE-RSP COMMAND
// =====================

var createRspCommand = new Command("create-rsp", "Create a response file for bundle command");

createRspCommand.SetHandler(() =>
{
    Console.Write("Languages (or 'all'): ");
    var languages = Console.ReadLine();

    Console.Write("Output file name: ");
    var output = Console.ReadLine();

    Console.Write("Add source note? (y/n): ");
    var note = Console.ReadLine()?.ToLower() == "y";

    Console.Write("Sort by (name/type): ");
    var sort = Console.ReadLine();

    Console.Write("Remove empty lines? (y/n): ");
    var remove = Console.ReadLine()?.ToLower() == "y";

    Console.Write("Author name (optional): ");
    var author = Console.ReadLine();

    var command = new StringBuilder("bundle ");

    command.Append($"--language {languages} ");
    command.Append($"--output {output} ");

    if (note)
        command.Append("--note ");

    if (!string.IsNullOrWhiteSpace(sort))
        command.Append($"--sort {sort} ");

    if (remove)
        command.Append("--remove-empty-lines ");

    if (!string.IsNullOrWhiteSpace(author))
        command.Append($"--author \"{author}\" ");

    File.WriteAllText("bundle.rsp", command.ToString());

    Console.WriteLine("Response file 'bundle.rsp' created successfully!");
});

rootCommand.AddCommand(createRspCommand);

// =====================

await rootCommand.InvokeAsync(args);
